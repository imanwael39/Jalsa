// core/services/chat.service.ts import { Injectable, inject, signal, effect } from '@angular/core'; import { HttpClientService } from '../api/http-client.service'; import { API } from '../api/api-endpoints'; import { ChatMessage, SendChatMessageRequest, ChatSession, CrisisAlert } from '../models'; import { AuthService } from './auth.service'; import { environment } from '../../../environments/environment'; import { HubConnection, HubConnectionBuilder, LogLevel } from '@microsoft/signalr'; import { Observable } from 'rxjs'; @Injectable({ providedIn: 'root', }) export class ChatService { private http = inject(HttpClientService); private authService = inject(AuthService); private connection: HubConnection | null = null; private isConnected = signal(false); // Public signals for message updates private messagesSignal = signal<ChatMessage[]>([]); readonly messages = this.messagesSignal.asReadonly(); readonly connected = this.isConnected.asReadonly(); constructor() { // Initialize connection when user is authenticated effect(() => { const user = this.authService.currentUser(); if (user) { this.initializeConnection(); } else { this.stopConnection(); } }); } private initializeConnection() { const token = this.authService.getToken(); if (!token) return; this.connection = new HubConnectionBuilder() .withUrl(environment.signalRHubUrl, { accessTokenFactory: () => token, }) .configureLogging(LogLevel.Information) .withAutomaticReconnect([0, 2000, 5000, 10000, 30000]) .build(); // Register event handlers this.connection.on('ReceiveMessage', (message: ChatMessage) => { this.messagesSignal.update(list => [...list, message]); }); this.connection.on('CrisisAlert', (alert: CrisisAlert) => { // Handle crisis alert (show notification, etc.) console.warn('Crisis alert received:', alert); }); this.connection.start() .then(() => { this.isConnected.set(true); console.log('SignalR connected'); }) .catch(err => { console.error('SignalR connection failed:', err); this.isConnected.set(false); }); } private stopConnection() { if (this.connection) { this.connection.stop().then(() => { this.isConnected.set(false); }); } } // Public API methods sendMessage(sessionId: string, message: string): Promise { if (!this.connection) { return Promise.reject('Connection not established'); } return this.connection.invoke('SendMessage', { sessionId, message }); } sendCrisisAlert(sessionId: string, message: string): Promise { if (!this.connection) { return Promise.reject('Connection not established'); } return this.connection.invoke('SendCrisisAlert', { sessionId, message }); } getHistory(sessionId: string): Observable<ChatMessage[]> { return this.http.get<ChatMessage[]>(API.chat.history(sessionId)); } getActiveSessions(): Observable<ChatSession[]> { return this.http.get<ChatSession[]>(API.chat.activeSessions); } getCrisisAlerts(): Observable<CrisisAlert[]> { return this.http.get<CrisisAlert[]>(API.chat.crisisAlerts); } resolveCrisisAlert(alertId: string): Observable<void> { return this.http.post<void>(API.chat.resolveCrisis(alertId), null); } // Cleanup dispose() { this.stopConnection(); } }
