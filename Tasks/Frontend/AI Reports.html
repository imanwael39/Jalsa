// core/services/report.service.ts import { Injectable, inject } from '@angular/core'; import { Observable } from 'rxjs'; import { HttpClientService } from '../api/http-client.service'; import { API } from '../api/api-endpoints'; import { Report, GenerateReportRequest, UpdateReportRequest, ReportListResponse } from '../models'; @Injectable({ providedIn: 'root', }) export class ReportService { private http = inject(HttpClientService); getReports(params: { patientId?: string; status?: string; page?: number; size?: number }): Observable<ReportListResponse> { return this.http.get<ReportListResponse>(API.reports.base, { params }); } getReport(id: string): Observable<Report> { return this.http.get<Report>(API.reports.byId(id)); } generateReport(data: GenerateReportRequest): Observable<Report> { return this.http.post<Report>(API.reports.generate, data); } updateReport(id: string, data: UpdateReportRequest): Observable<Report> { return this.http.put<Report>(API.reports.byId(id), data); } approveReport(id: string): Observable<void> { return this.http.post<void>(API.reports.approve(id), null); } rejectReport(id: string): Observable<void> { return this.http.post<void>(API.reports.reject(id), null); } exportPDF(id: string): Observable<Blob> { return this.http.get<Blob>(API.reports.export(id), { responseType: 'blob' }); } }

// core/state/report-state.service.ts import { Injectable, signal } from '@angular/core'; import { Report } from '../models'; @Injectable({ providedIn: 'root', }) export class ReportStateService { private reportsSignal = signal<Report[]>([]); private selectedReportSignal = signal<Report | null>(null); private totalCountSignal = signal(0); private loadingSignal = signal(false); private generatingSignal = signal(false); private errorSignal = signal<string | null>(null); readonly reports = this.reportsSignal.asReadonly(); readonly selectedReport = this.selectedReportSignal.asReadonly(); readonly totalCount = this.totalCountSignal.asReadonly(); readonly loading = this.loadingSignal.asReadonly(); readonly generating = this.generatingSignal.asReadonly(); readonly error = this.errorSignal.asReadonly(); setReports(reports: Report[], totalCount: number) { this.reportsSignal.set(reports); this.totalCountSignal.set(totalCount); this.errorSignal.set(null); } addReport(report: Report) { this.reportsSignal.update(list => [report, ...list]); this.totalCountSignal.update(count => count + 1); } updateReport(updated: Report) { this.reportsSignal.update(list => list.map(r => r.id === updated.id ? updated : r)); if (this.selectedReportSignal()?.id === updated.id) { this.selectedReportSignal.set(updated); } } removeReport(id: string) { this.reportsSignal.update(list => list.filter(r => r.id !== id)); this.totalCountSignal.update(count => count - 1); } selectReport(report: Report) { this.selectedReportSignal.set(report); } clearSelected() { this.selectedReportSignal.set(null); } setLoading(loading: boolean) { this.loadingSignal.set(loading); } setGenerating(generating: boolean) { this.generatingSignal.set(generating); } setError(error: string | null) { this.errorSignal.set(error); } reset() { this.reportsSignal.set([]); this.selectedReportSignal.set(null); this.totalCountSignal.set(0); this.loadingSignal.set(false); this.generatingSignal.set(false); this.errorSignal.set(null); } }
